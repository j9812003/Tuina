function setupDateLimits() { const dateInput = document.getElementById('date'); // 取得台灣時間 (UTC+8) 的 YYYY-MM-DD 字串 const getTaiwanDateStr = (dateObj) => { const formatter = new Intl.DateTimeFormat('en-CA', { timeZone: 'Asia/Taipei', year: 'numeric', month: '2-digit', day: '2-digit' }); return formatter.format(dateObj); // 會回傳 "YYYY-MM-DD" }; const now = new Date(); // 1. 台灣時間的今天 const todayStr = getTaiwanDateStr(now); // 2. 台灣時間的 30 天後 const maxDate = new Date(now.getTime() + (30 * 24 * 60 * 60 * 1000)); const maxStr = getTaiwanDateStr(maxDate); // 3. 綁定至 HTML date input dateInput.min = todayStr; dateInput.max = maxStr; }
